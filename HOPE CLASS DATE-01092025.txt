PROBLEM 2: REPLACE EACH ELEMNET IN ARRAY WITH SUM OF ALL OTHER ELEMENTS EXCEPT ITSELF

Input:  arr = [1, 2, 3, 4]
Output: arr = [9, 8, 7, 6]
Explanation:
For index 0 → 2+3+4 = 9  
For index 1 → 1+3+4 = 8  
For index 2 → 1+2+4 = 7  
For index 3 → 1+2+3 = 6

so we will find the sum of them and replace them by subtracting the remaining

    }

        // Input array elements
        
        for (int i = 0; i < n; i++) {

        }
	int front , rear,temp;
	for(front=0,rear=n-1;front<rear ; front++ , rear--){

import java.util.Scanner;
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        // Take array size from user
        int n = sc.nextInt();

        int[] arr = new int[n];
        

        // Input array elements
        
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
            
        }
	int f,r;
	for (f=0,r=n-1;f<r;f++,r--) {
	    while(arr[f]==0){
		f++;
		}
	    while(arr[r]==1){
		r--;
		}
	    if(f<r){
		int t=arr[f];
		arr[f]=arr[r];
		arr[r]=t;
		}	
		
        }

        // Print the modified array

#include <stdio.h>

int main() {
    int num;
    scanf("%d", &num);

    int even = 0, odd = 0;
    int even_place = 1, odd_place = 1;
    int temp = num;

    // Process each digit
    while (temp > 0) {
        int d = temp % 10; // Get last digit

        if (d % 2 == 0) {
            even = even + d * even_place;  // Place even digits in correct order
            even_place *= 10;
        } else {
            odd = odd + d * odd_place;     // Place odd digits in correct order
            odd_place *= 10;
        }
        temp /= 10;
    }

    // Combine even digits first, then odd digits
    int result = even * odd_place + odd;
    printf("%d\n", result);

    return 0;
}
